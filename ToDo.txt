private void generateDungeon() {
        List<Tile> tiles = new ArrayList<>();

        // Core content
        for (int i = 0; i < 5; i++) tiles.add(new Tile(TileType.GOLD));
        for (int i = 0; i < 5; i++) tiles.add(new Tile(TileType.ENEMY));
        for (int i = 0; i < 2; i++) tiles.add(new Tile(TileType.TRAP_FIRE));
        for (int i = 0; i < 2; i++) tiles.add(new Tile(TileType.TRAP_POISON));
        tiles.add(new Tile(TileType.TRAP_ACID));
        tiles.add(new Tile(TileType.TRAP_FREEZE));
        tiles.add(new Tile(TileType.TRAP_PITFALL));

        // Add one up stair if not first floor
        if (currentFloor > 1) tiles.add(new Tile(TileType.STAIR_UP));

        // Always add one down stair, sometimes locked
        TileType placedLockedStair = null;
        if (Math.random() < 0.5) {
            placedLockedStair = Math.random() < 0.5 ? TileType.STAIR_DOWN_LOCKED_RED : TileType.STAIR_DOWN_LOCKED_BLUE;
            tiles.add(new Tile(placedLockedStair));
        } else {
            tiles.add(new Tile(TileType.STAIR_DOWN));
        }

        // If locked stair was added, add the matching key
        if (placedLockedStair != null) {
            if (placedLockedStair == TileType.STAIR_DOWN_LOCKED_RED) {
                tiles.add(new Tile(TileType.RED_KEY));
            } else if (placedLockedStair == TileType.STAIR_DOWN_LOCKED_BLUE) {
                tiles.add(new Tile(TileType.BLUE_KEY));
            }
        }

        // Fill remaining tiles
        while (tiles.size() < GRID_SIZE * GRID_SIZE) {
            tiles.add(new Tile(TileType.EMPTY));
        }

        Collections.shuffle(tiles);

        int index = 0;
        for (int row = 0; row < GRID_SIZE; row++) {
            for (int col = 0; col < GRID_SIZE; col++) {
                dungeonGrid[row][col] = tiles.get(index++);
            }
        }
    }
